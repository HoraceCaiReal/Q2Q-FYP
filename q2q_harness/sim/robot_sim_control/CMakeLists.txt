cmake_minimum_required(VERSION 3.8)
project(robot_sim_control)
find_package(ament_cmake REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(ignition-msgs8 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_library(q2q_kinematic_carry SHARED src/kinematic_carry.cpp)
target_compile_features(q2q_kinematic_carry PUBLIC cxx_std_17)
target_link_libraries(q2q_kinematic_carry ignition-gazebo6::core ignition-transport11::core ignition-plugin1::register)
install(TARGETS q2q_kinematic_carry LIBRARY DESTINATION lib)
add_executable(carry_pose_client src/carry_pose_client.cpp)
target_compile_features(carry_pose_client PUBLIC cxx_std_17)
target_link_libraries(carry_pose_client ignition-transport11::core ignition-msgs8::core)
install(TARGETS carry_pose_client DESTINATION lib/${PROJECT_NAME})
add_library(q2q_gain_system SHARED src/gain_system.cpp)
target_compile_features(q2q_gain_system PUBLIC cxx_std_17)
ament_target_dependencies(q2q_gain_system gz_ros2_control rclcpp pluginlib hardware_interface rclcpp_lifecycle)
target_link_libraries(q2q_gain_system ignition-gazebo6::core)
pluginlib_export_plugin_description_file(gz_ros2_control robot_sim_control_plugins.xml)
install(TARGETS q2q_gain_system LIBRARY DESTINATION lib)
ament_package()
